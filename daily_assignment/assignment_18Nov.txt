Code:
package dailyAssignment3;
import java.util.*;


class Employee5{
	String name;
	double salary;
	public Employee5(String name, double salary) {
		this.name = name;
		this.salary = salary;
	}
	public String getName() {
		return name;
	}
	public void setName(String name) {
		this.name = name;
	}
	public double getSalary() {
		return salary;
	}
	public void setSalary(double salary) {
		this.salary = salary;
	}
	
}

public class Program_JavaStream2 {

	public static void main(String[] args) {
		// TODO Auto-generated method stub
				
				Employee5 obj1= new Employee5("Rahul",50000);
				Employee5 obj2= new Employee5("Priya",70000);
				Employee5 obj3= new Employee5("Amit",60000);
				
				ArrayList<Employee5>list = new ArrayList<Employee5>();
				
				list.add(obj1);
				list.add(obj2);
				list.add(obj3);
				
				System.out.println("Ascending order based on salary");
				list.stream().sorted(Comparator.comparingDouble(Employee5::getSalary)).
				forEach(e->System.out.println(e.name +" " + e.salary));
				
				System.out.println("\nDescending order based on salary");
				list.stream().sorted(Comparator.comparingDouble(Employee5::getSalary).
				reversed()).
				forEach(e->System.out.println(e.name +" " + e.salary));
				
				double totalsal = list.stream().mapToDouble(Employee5::getSalary).sum();
				System.out.println("\nTotal Salary is :" +totalsal);

	}

}
