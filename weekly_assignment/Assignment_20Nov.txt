package WeeklyAssignment2;
import java.util.*;
import java.util.stream.Collectors;
import java.util.stream.*;


class Employee {
    private String name;
    private String department;
    private int salary;

    public Employee(String name, String department, int salary) {
        this.name = name;
        this.department = department;
        this.salary = salary;
    }

    public String getName() {
        return name;
    }

    public String getDepartment() {
        return department;
    }

    public int getSalary() {
        return salary;
    }

    @Override
    public String toString() {
        return name + " - " + department + " - " + salary;
    }
}


public class Program_JavaStreamAPIAssignment {

	public static void main(String[] args) {
		List<Employee> employees = Arrays.asList(
			    new Employee("Rahul", "IT", 60000),
			    new Employee("Priya", "HR", 45000),
			    new Employee("Amit", "IT", 75000),
			    new Employee("Neha", "Finance", 55000),
			    new Employee("Karan", "IT", 60000),
			    new Employee("Simran", "HR", 50000),
			    new Employee("Rohit", "Finance", 75000),
			    new Employee("Ankit", "IT", 40000)
			);
		
		System.out.println("Employess whose salary is greater than 50000");
		employees.stream()
		.filter(s->s.getSalary()>50000)
		.forEach(s->System.out.println(s.getName()));
		
		//Find all employees working in the IT department and display only their names.
		
		System.out.println("Employees who all are working in the IT department");
		employees.stream()
		.filter(s->s.getDepartment().equals("IT"))
		.map(Employee::getName)
		.forEach(s->System.out.println(s));
		
		//Sort all employees by salary in ascending order and print employee names along with their salaries.
		System.out.println("Sorting All Employees in Ascending Order");
		employees.stream()
		.sorted(Comparator.comparingInt(Employee::getSalary))
		.forEach(s->System.out.println(s.getName()+"-"+s.getSalary()));
		
		
		
		//Distinct Salaries in ascending order
		System.out.println("Distinct Salaries in ascending order");
		employees.stream()
				.map(Employee::getSalary)
				.distinct().sorted()
				.forEach(s->System.out.println(s));
		
		//Distinct Department Names in which employees has salary greater than or equal to 50000
		System.out.println("Distinct Department Names in which employees has salary greater than or equal to 50000");
		employees.stream()
		.filter(s->s.getSalary()>=50000)
		.map(Employee::getDepartment)
		.distinct().sorted()
		.forEach(s->System.out.println(s));
		
	



	}

}
